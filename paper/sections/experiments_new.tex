\section{Experiments}
\label{sec:experiments}

This section presents the experimental evaluation of our framework for generating
privacy-preserving explanations for semantic differences between query outputs.

\subsection{Experimental Setup}

\subsubsection{Datasets}

\paragraph{IPUMS-CPS.}
We use a down-sampled version of the IPUMS-CPS dataset containing
\textbf{100,000} tuples.

\paragraph{Stack Overflow Developer Survey.}
We also evaluate the method on the Stack Overflow Developer Survey dataset.

\paragraph{Hospital Dataset.}
To evaluate the framework in the context of consistent query answering (CQA),
we use the Hospital dataset commonly used in prior repair and data-cleaning
work \cite{rekatsinas2017}.

The dataset contains hospital records with attributes such as:
\[
\texttt{HospitalName},\ \texttt{City},\ \texttt{State},
\texttt{ZipCode},\ \texttt{PhoneNumber},\ \texttt{HospitalType}.
\]

The dataset contains inconsistencies with respect to functional dependencies:
\[
\texttt{HospitalName} \rightarrow \texttt{ZipCode},
\]
\[
\texttt{HospitalName} \rightarrow \texttt{PhoneNumber},
\]
and:
\[
\texttt{ZipCode} \rightarrow \texttt{City}.
\]

\subsection{Hospital Repair Case Study}

\paragraph{ZipCode Consistency Query Pair.}

\begin{verbatim}
Q1:
SELECT *
FROM Hospital
WHERE State = 'CA'
  AND HospitalType = 'Acute Care';

Q2:
SELECT *
FROM Hospital h
WHERE State = 'CA'
  AND HospitalType = 'Acute Care'
  AND NOT EXISTS (
      SELECT *
      FROM Hospital h2
      WHERE h.HospitalName = h2.HospitalName
        AND h.ZipCode <> h2.ZipCode
  );
\end{verbatim}

\paragraph{Phone Number Consistency Query Pair.}

\begin{verbatim}
Q1:
SELECT *
FROM Hospital
WHERE City = 'Chicago';

Q2:
SELECT *
FROM Hospital h
WHERE City = 'Chicago'
  AND NOT EXISTS (
      SELECT *
      FROM Hospital h2
      WHERE h.HospitalName = h2.HospitalName
        AND h.PhoneNumber <> h2.PhoneNumber
  );
\end{verbatim}

\paragraph{City Consistency Query Pair.}

\begin{verbatim}
Q1:
SELECT *
FROM Hospital
WHERE HospitalType = 'General';

Q2:
SELECT *
FROM Hospital h
WHERE HospitalType = 'General'
  AND NOT EXISTS (
      SELECT *
      FROM Hospital h2
      WHERE h.ZipCode = h2.ZipCode
        AND h.City <> h2.City
  );
\end{verbatim}
